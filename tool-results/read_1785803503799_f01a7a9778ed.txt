    41→  useEffect(() => {
    42→    const onKey = (e: KeyboardEvent) => {
    43→      if (e.ctrlKey && e.shiftKey && (e.key === "A" || e.key === "a")) {
    44→        e.preventDefault();
    45→        setOpen((o) => !o);
    46→      }
    47→      if (e.key === "Escape") setOpen(false);

    51→    window.addEventListener("n2k-open-admin", onOpen);

    92→          <motion.div

    94→            animate={{ y: 0, opacity: 1 }}
    95→            exit={{ y: 30, opacity: 0 }}

   179→function LoginPanel({ onLoggedIn }: { onLoggedIn: () => void }) {
   180→  const [email, setEmail] = useState("admin@n2klabs.com");
   181→  const [password, setPassword] = useState("");
   182→  const [loading, setLoading] = useState(false);
   183→  const [err, setErr] = useState("");
   184→
   185→  const submit = async (e: React.FormEvent) => {
   186→    e.preventDefault();
   187→    setLoading(true);
   188→    setErr("");
   189→    try {
   190→      const res = await fetch("/api/auth/login", {
   191→        method: "POST",
   192→        headers: { "Content-Type": "application/json" },
   193→        body: JSON.stringify({ email, password }),
   194→      });
   195→      const data = await res.json();
   196→      if (!res.ok) throw new Error(data.error || "Login failed");
   197→      onLoggedIn();
   198→    } catch (e) {
   199→      setErr(e instanceof Error ? e.message : "Login failed");

   201→      setLoading(false);
   202→    }

   204→
   205→  return (
   206→    <div className="flex h-full items-center justify-center p-6">

   268→/* ============ shared hooks ============ */
   269→function useFetch<T>(url: string) {
   270→  const [data, setData] = useState<T | null>(null);
   271→  const [loading, setLoading] = useState(true);
   272→  const [err, setErr] = useState("");
   273→
   274→  const reload = useCallback(() => {
   275→    setLoading(true);
   276→    fetch(url)
   277→      .then((r) => r.json())
   278→      .then((d) => setData(d))
   279→      .catch((e) => setErr(e.message))
   280→      .finally(() => setLoading(false));
   281→  }, [url]);
   282→
   283→  useEffect(() => {
   284→    // eslint-disable-next-line react-hooks/set-state-in-effect
   285→    reload();
   286→  }, [reload]);
   287→
   288→  return { data, loading, err, reload, setData };
   289→}

   345→function ProjectsAdmin() {
   346→  const { data, loading, reload, setData } = useFetch<{ projects: Project[] }>("/api/projects");
   347→  const [editing, setEditing] = useState<Project | null>(null);
   348→  const [saving, setSaving] = useState(false);
   349→
   350→  const onSave = async (p: Project) => {
   351→    setSaving(true);
   352→    try {
   353→      const tags = typeof p.tags === "string" ? JSON.parse(p.tags || "[]") : p.tags || [];
   354→      const body = { ...p, tags };
   355→      const method = p.id ? "PUT" : "POST";
   356→      const res = await fetch("/api/projects", {
   357→        method,
   358→        headers: { "Content-Type": "application/json" },
   359→        body: JSON.stringify(body),
   360→      });
   361→      if (!res.ok) throw new Error("save failed");
   362→      setEditing(null);
   363→      reload();
   364→    } catch {
   365→      alert("Save failed");
   366→    } finally {
   367→      setSaving(false);
   368→    }
   369→  };
   370→
   371→  const onDelete = async (id: string) => {
   372→    if (!confirm("Delete this project?")) return;
   373→    await fetch("/api/projects", {
   374→      method: "DELETE",
   375→      headers: { "Content-Type": "application/json" },
   376→      body: JSON.stringify({ id }),
   377→    });
   378→    reload();
   379→  };
   380→
   381→  if (loading) return <Loader />;
   382→  const projects = data?.projects || [];
   383→
   384→  return (
   385→    <div>
   386→      <Toolbar title="Projects" onAdd={() => setEditing({} as Project)} />
   387→      {editing && (
   388→        <ProjectEditor
   389→          project={editing}
   390→          saving={saving}
   391→          onSave={onSave}
   392→          onCancel={() => setEditing(null)}
   393→        />
   394→      )}
   395→      <div className="space-y-3">
   396→        {projects.map((p) => (
   397→          <ItemCard key={p.id}>
   398→            <div className="flex items-center gap-4">
   399→              <div className="h-12 w-12 shrink-0 rounded-lg bg-[rgba(var(--accent-rgb),0.1)] border border-[rgba(var(--accent-rgb),0.2)] flex items-center justify-center overflow-hidden">
   400→                {p.image ? (
   401→                  <img src={p.image} alt="" className="h-full w-full object-cover" />
   402→                ) : (
   403→                  <FolderKanban className="h-5 w-5 text-[var(--accent)]" />
   404→                )}
   405→              </div>
   406→              <div className="flex-1 min-w-0">
   407→                <div className="flex items-center gap-2">
   408→                  <span className="font-semibold text-[#f0ece6] truncate">{p.title}</span>
   409→                  {p.featured && (
   410→                    <span className="rounded px-1.5 py-0.5 text-[9px] bg-[rgba(var(--accent-rgb),0.1)] text-[var(--accent)] font-mono">
   411→                      FEATURED
   412→                    </span>
   413→                  )}
   414→                </div>
   415→                <div className="mono-label text-[#9aa0a8] mt-0.5">
   416→                  {p.client} · {p.category}
   417→                </div>
   418→              </div>
   419→              <div className="flex gap-2">
   420→                <button
   421→                  onClick={() => setEditing(p)}
   422→                  className="rounded-lg border border-white/10 px-3 py-1.5 text-xs text-[#8a8a93] hover:border-[rgba(var(--accent-rgb),0.5)] hover:text-[var(--accent)]"
   423→                >
   424→                  Edit
   425→                </button>
   426→                <button
   427→                  onClick={() => onDelete(p.id)}
   428→                  className="rounded-lg border border-white/10 p-1.5 text-[#8a8a93] hover:border-[#ff4d5e]/50 hover:text-[#ff4d5e]"
   429→                >
   430→                  <Trash2 className="h-3.5 w-3.5" />
   431→                </button>
   432→              </div>
   433→            </div>
   434→          </ItemCard>
   435→        ))}
   436→        {projects.length === 0 && <Empty label="No projects yet" />}
   437→      </div>
   438→    </div>
   439→  );
   440→}
   441→
   442→function ProjectEditor({
   443→  project,
   444→  saving,
   445→  onSave,
   446→  onCancel,
   447→}: {
   448→  project: Project;
   449→  saving: boolean;
   450→  onSave: (p: Project) => void;
   451→  onCancel: () => void;
   452→}) {
   453→  const [f, setF] = useState<Project>(project);
   454→  const set = (k: keyof Project, v: unknown) => setF((p) => ({ ...p, [k]: v }));
   455→  const tagsStr =
   456→    typeof f.tags === "string"
   457→      ? (() => {
   458→          try {
   459→            return JSON.parse(f.tags || "[]").join(", ");
   460→          } catch {
   461→            return "";
   462→          }
   463→        })()
   464→      : "";
   465→
   466→  const upload = async (file: File) => {
   467→    const fd = new FormData();
   468→    fd.append("file", file);
   469→    const res = await fetch("/api/upload", { method: "POST", body: fd });
   470→    const d = await res.json();
   471→    if (d.url) set("image", d.url);
   472→  };
   473→
   474→  return (
   475→    <div className="mb-6 rounded-xl border border-[rgba(var(--accent-rgb),0.3)] bg-[#0a0a0c]/50 p-5">
   476→      <div className="mono-label text-[var(--accent)] mb-4">
   477→        {project.id ? "EDIT PROJECT" : "NEW PROJECT"}
   478→      </div>
   479→      <div className="grid sm:grid-cols-2 gap-3">
   480→        <Field label="Title"><input className={inputCls} value={f.title || ""} onChange={(e) => set("title", e.target.value)} /></Field>
   481→        <Field label="Client"><input className={inputCls} value={f.client || ""} onChange={(e) => set("client", e.target.value)} /></Field>
   482→        <Field label="Category"><input className={inputCls} value={f.category || ""} onChange={(e) => set("category", e.target.value)} placeholder="Web Development" /></Field>
   483→        <Field label="URL"><input className={inputCls} value={f.url || ""} onChange={(e) => set("url", e.target.value)} placeholder="https://..." /></Field>
   484→        <Field label="Order"><input className={inputCls} type="number" value={f.order || 0} onChange={(e) => set("order", Number(e.target.value))} /></Field>
   485→        <Field label="Featured">
   486→          <select className={inputCls} value={f.featured ? "1" : "0"} onChange={(e) => set("featured", e.target.value === "1")}>
   487→            <option value="0">No</option>
   488→            <option value="1">Yes</option>
   489→          </select>
   490→        </Field>
   491→      </div>
   492→      <div className="mt-3 space-y-3">
   493→        <Field label="Description"><textarea className={inputCls} rows={2} value={f.description || ""} onChange={(e) => set("description", e.target.value)} /></Field>
   494→        <Field label="Challenge"><textarea className={inputCls} rows={2} value={f.challenge || ""} onChange={(e) => set("challenge", e.target.value)} /></Field>
   495→        <Field label="Solution"><textarea className={inputCls} rows={2} value={f.solution || ""} onChange={(e) => set("solution", e.target.value)} /></Field>
   496→        <Field label="Result"><textarea className={inputCls} rows={2} value={f.result || ""} onChange={(e) => set("result", e.target.value)} /></Field>
   497→        <Field label="Tags (comma separated)"><input className={inputCls} defaultValue={tagsStr} onBlur={(e) => set("tags", JSON.stringify(e.target.value.split(",").map((s) => s.trim()).filter(Boolean)))} /></Field>
   498→        <Field label="Image">
   499→          <div className="flex items-center gap-3">
   500→            <input type="file" accept="image/*" onChange={(e) => e.target.files?.[0] && upload(e.target.files[0])} className="hidden" id="proj-img" />
   501→            <label htmlFor="proj-img" className="cursor-pointer flex items-center gap-2 rounded-lg border border-white/10 px-3 py-2 text-xs text-[#8a8a93] hover:border-[rgba(var(--accent-rgb),0.5)] hover:text-[var(--accent)]">
   502→              <Upload className="h-3.5 w-3.5" /> Upload
   503→            </label>
   504→            {f.image && <span className="text-xs text-[#8a8a93] truncate">{f.image}</span>}
   505→            {f.image && (
   506→              <button onClick={() => set("image", null)} className="text-xs text-[#ff4d5e]">remove</button>
   507→            )}
   508→          </div>
   509→        </Field>

   804→/* ============ PASSWORD ============ */
   805→function PasswordAdmin() {
   806→  const [cur, setCur] = useState("");
   807→  const [next, setNext] = useState("");
   808→  const [loading, setLoading] = useState(false);
   809→  const [msg, setMsg] = useState<{ ok: boolean; text: string } | null>(null);
   810→
   811→  const submit = async (e: React.FormEvent) => {
   812→    e.preventDefault();
   813→    setLoading(true);
   814→    setMsg(null);
   815→    try {
   816→      const res = await fetch("/api/admin/password", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ currentPassword: cur, newPassword: next }) });
   817→      const d = await res.json();
   818→      if (!res.ok) throw new Error(d.error || "failed");
   819→      setMsg({ ok: true, text: "Password updated successfully." });
   820→      setCur(""); setNext("");
   821→    } catch (e) {
   822→      setMsg({ ok: false, text: e instanceof Error ? e.message : "Failed" });
   823→    } finally {
   824→      setLoading(false);
   825→    }
   826→  };
   827→
   828→  return (
   829→    <div>
   830→      <Toolbar title="Change Password" />
   831→      <form onSubmit={submit} className="space-y-4 max-w-sm">
   832→        <Field label="Current Password"><input type="password" className={inputCls} value={cur} onChange={(e) => setCur(e.target.value)} required /></Field>
   833→        <Field label="New Password"><input type="password" className={inputCls} value={next} onChange={(e) => setNext(e.target.value)} required /></Field>
   834→        {msg && <p className={`text-sm ${msg.ok ? "text-[var(--accent)]" : "text-[#ff4d5e]"}`}>{msg.text}</p>}
   835→        <button type="submit" disabled={loading} className="flex items-center gap-2 rounded-full bg-[#f0ece6] px-5 py-2.5 text-xs font-semibold text-[#0a0a0c] disabled:opacity-60">
   836→          {loading ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <KeyRound className="h-3.5 w-3.5" />} Update Password
   837→        </button>
   838→      </form>
   839→    </div>
   840→  );
   841→}
